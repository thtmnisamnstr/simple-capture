# Camera bubble (M2): opens with the recorder panel, shows live camera frames where the camera lands
# for a whole-display Original frame, and follows a drag; a 9:16 frame brings up the stage preview
# with live frames and leaves the bubble floating.
cmd set_settings {"settings":{"source":{"kind":"display","display_id":"@DISPLAY@"},"camera":"@CAMERA@","mic":null,"bubble_shown":true}}
show panel
wait 2500
cmd bubble_state
log-window-rects
wait 1000
cmd bubble_state
cmd bubble_drag {"phase":"begin","x":1000,"y":500,"corner":null}
cmd bubble_drag {"phase":"move","x":900,"y":450,"corner":null}
cmd bubble_drag {"phase":"end","x":0,"y":0,"corner":null}
wait 500
cmd bubble_state
log-window-rects
cmd set_settings {"settings":{"source":{"kind":"display","display_id":"@DISPLAY@"},"camera":"@CAMERA@","mic":null,"bubble_shown":true,"canvas":{"aspect":"9:16","background":{"kind":"solid","color":[0.1,0.2,0.3,1]},"background_image":null}}}
wait 2500
cmd bubble_state
log-window-rects
wait 1000
cmd bubble_state
quit
