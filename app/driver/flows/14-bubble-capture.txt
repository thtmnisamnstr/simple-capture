# Camera bubble during a display recording (M2): the bubble never appears in the screen track,
# and dragging it while recording writes camera keyframes.
cmd set_settings {"settings":{"source":{"kind":"display","display_id":"@DISPLAY@"},"camera":"@CAMERA@","mic":null,"bubble_shown":false}}
show panel
wait 1500
cmd start_recording {"request":{"source":{"kind":"display","display_id":"@DISPLAY@"},"via":"own_chooser","title":"drv bubble capture","settings":{"camera":"@CAMERA@","bubble_shown":false}}}
wait 2500
cmd set_settings {"settings":{"source":{"kind":"display","display_id":"@DISPLAY@"},"camera":"@CAMERA@","mic":null,"bubble_shown":true}}
wait 500
log-window-rects
cmd bubble_state
wait 2000
cmd bubble_drag {"phase":"begin","x":1000,"y":500,"corner":null}
cmd bubble_drag {"phase":"move","x":950,"y":480,"corner":null}
wait 100
cmd bubble_drag {"phase":"move","x":900,"y":460,"corner":null}
cmd bubble_drag {"phase":"end","x":0,"y":0,"corner":null}
wait 1500
cmd stop_recording
wait-phase saved 30000
quit
